function ReceiveTab({products,setProducts,receives,setReceives,movements,setMovements,prodHistory,setProdHistory,showToast,save,currentUser}) {

  const ef=()=>({productId:"",sourceType:"ร้านค้าทั่วไป",sourceName:"",qty:"",price:"",buyer:currentUser.displayName,date:todayStr(),time:nowTime(),note:"",attachImages:[]});

  const [form,setForm]=useState(ef());

  const [editId,setEditId]=useState(null);

  const [viewRec,setViewRec]=useState(null);

  const [viewer,setViewer]=useState(null);


  const savePH=async h=>{try{await window.storage?.set("ims6_prodhistory",JSON.stringify(h));}catch(e){}};

  const resetForm=()=>{setForm(ef());setEditId(null);};


  const handleSubmit=()=>{

    if(!form.productId||!form.qty||!form.buyer){showToast("กรุณากรอกข้อมูลให้ครบ","error");return;}

    const prod=products.find(p=>p.id===Number(form.productId));

    const qn=Number(form.qty),pn=Number(form.price)||0;

    if(editId){

      const old=receives.find(r=>r.id===editId);

      const diff=qn-(old.qty||0);

      const updP=products.map(p=>p.id===Number(form.productId)?{...p,qty:p.qty+diff}:p);

      const newR=receives.map(r=>r.id===editId?{...r,...form,qty:qn,price:pn,totalPrice:qn*pn,productCode:prod.code,productName:prod.name}:r);

      setProducts(updP);setReceives(newR);save(updP,newR);showToast("แก้ไขรายการเรียบร้อย");

    }else{

      const rec={id:Date.now(),productId:Number(form.productId),productCode:prod.code,productName:prod.name,sourceType:form.sourceType,sourceName:form.sourceName,qty:qn,price:pn,totalPrice:qn*pn,buyer:form.buyer,date:form.date,time:form.time,note:form.note,attachImages:form.attachImages||[]};

      const updP=products.map(p=>p.id===Number(form.productId)?{...p,qty:p.qty+qn}:p);

      const newR=[rec,...receives];

      const newM=[{id:Date.now(),type:"รับเข้า",productCode:prod.code,productName:prod.name,qty:qn,by:form.buyer,date:form.date,time:form.time},...movements];

      const entry={id:rec.id,price:pn,sourceType:form.sourceType,sourceName:form.sourceName,buyer:form.buyer,date:form.date,qty:qn};

      const nh={...prodHistory,[prod.id]:[entry,...(prodHistory[prod.id]||[])]};

      setProdHistory(nh);savePH(nh);

      setProducts(updP);setReceives(newR);setMovements(newM);save(updP,newR,undefined,newM);

      showToast(`รับสินค้า "${prod.name}" จำนวน ${qn} เรียบร้อย`);

    }

    resetForm();

  };


  const handleEdit=r=>{setForm({...r,productId:r.productId,attachImages:r.attachImages||[]});setEditId(r.id);};

  const handleDel=id=>{

    if(!window.confirm("ยืนยันลบ?"))return;

    const old=receives.find(r=>r.id===id);

    const updP=products.map(p=>p.id===old.productId?{...p,qty:p.qty-(old.qty||0)}:p);

    const newR=receives.filter(r=>r.id!==id);

    setProducts(updP);setReceives(newR);save(updP,newR);showToast("ลบรายการเรียบร้อย");

  };


  return (

    <div>

function IssueTab({products,setProducts,issues,setIssues,movements,setMovements,showToast,save,currentUser}) {

  const ef=()=>({productId:"",qty:"",requester:currentUser.displayName,department:"",date:todayStr(),time:nowTime(),note:"",attachImages:[]});

  const [form,setForm]=useState(ef());

  const [editId,setEditId]=useState(null);

  const [viewIss,setViewIss]=useState(null);

  const [viewer,setViewer]=useState(null);

  const resetForm=()=>{setForm(ef());setEditId(null);};


  const handleSubmit=()=>{

    if(!form.productId||!form.qty||!form.requester){showToast("กรุณากรอกข้อมูลให้ครบ","error");return;}

    const prod=products.find(p=>p.id===Number(form.productId));

    const qn=Number(form.qty);

    if(editId){

      const old=issues.find(i=>i.id===editId);

      const diff=qn-(old.qty||0);

      if(prod.qty-diff<0){showToast("สินค้าไม่เพียงพอ","error");return;}

      const updP=products.map(p=>p.id===Number(form.productId)?{...p,qty:p.qty-diff}:p);

      const newI=issues.map(i=>i.id===editId?{...i,...form,qty:qn}:i);

      setProducts(updP);setIssues(newI);save(updP,undefined,newI);showToast("แก้ไขรายการเรียบร้อย");

    }else{

      if(qn>prod.qty){showToast(`สินค้าไม่พอ เหลือ ${prod.qty} ${prod.unit}`,"error");return;}

      const iss={id:Date.now(),productId:Number(form.productId),productCode:prod.code,productName:prod.name,qty:qn,requester:form.requester,department:form.department,date:form.date,time:form.time,note:form.note,attachImages:form.attachImages||[]};

      const updP=products.map(p=>p.id===Number(form.productId)?{...p,qty:p.qty-qn}:p);

    }

          <div style={{marginBottom:8}}><Lbl>สินค้า</Lbl><select value={form.productId} onChange={e=>setForm(f=>({...f,productId:e.target.value}))} style={s()}><option value="">-- เลือกสินค้า --</option>{products.map(p=><option key={p.id} value={p.id}>{p.code} — {p.name} (เหลือ {p.qty} {p.unit})</option>)}</select></div>

            <div key={k} style={{marginBottom:8}}><Lbl>{l}</Lbl><input type={t} value={form[k]} onChange={e=>setForm(f=>({...f,[k]:e.target.value}))} style={s()}/></div>

          ))}

          <div style={{marginBottom:8}}><Lbl>แผนก</Lbl><select value={form.department} onChange={e=>setForm(f=>({...f,department:e.target.value}))} style={s()}><option value="">-- เลือกแผนก --</option>{DEPARTMENTS.map(d=><option key={d}>{d}</option>)}</select></div>

    </div>

  );

}
